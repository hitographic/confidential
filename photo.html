function triggerPhotoUpload(index) { currentCropIndex = index; document.getElementById(`photoInput_${index}`).click(); } function handlePhotoSelected(event, index) { const file = event.target.files[0]; if (!file) return; event.target.value = ''; const reader = new FileReader(); reader.onload = function(e) { const img = new Image(); img.onload = function() { openCropModal(img, index); }; img.src = e.target.result; }; reader.readAsDataURL(file); } function openCropModal(img, index) { currentCropIndex = index; currentCropData = img; document.getElementById('cropCaption').value = ''; const canvas = document.getElementById('cropCanvas'); const ctx = canvas.getContext('2d'); const size = Math.min(img.width, img.height); const sx = (img.width - size) / 2; const sy = (img.height - size) / 2; canvas.width = 400; canvas.height = 400; ctx.drawImage(img, sx, sy, size, size, 0, 0, 400, 400); showModal('cropModal'); } function confirmCrop() { const canvas = document.getElementById('cropCanvas'); const caption = document.getElementById('cropCaption').value.trim(); const dataUrl = canvas.toDataURL('image/png'); const index = currentCropIndex; if (!formPhotos[index]) formPhotos[index] = []; const photoObj = { id: 'photo_' + Date.now() + '_' + Math.random().toString(36).substr(2, 5), url: dataUrl, fileId: '', caption: caption, local: true }; formPhotos[index].push(photoObj); renderPhotosInForm(index); closeModal('cropModal'); currentCropIndex = null; currentCropData = null; } function cancelCrop() { closeModal('cropModal'); currentCropIndex = null; currentCropData = null; } function renderPhotosInForm(index) { const area = document.getElementById(`photoArea_${index}`); if (!area) return; const photos = formPhotos[index] || []; area.innerHTML = ''; photos.forEach((photo, pIdx) => { const thumb = document.createElement('div'); thumb.className = 'photo-thumb'; thumb.innerHTML = `
`; area.appendChild(thumb); }); const addBtn = document.createElement('div'); addBtn.className = 'photo-add-btn'; addBtn.title = 'Upload/Take Foto'; addBtn.innerHTML = ''; addBtn.onclick = () => triggerPhotoUpload(index); area.appendChild(addBtn); } function editPhotoCaption(itemIndex, photoIndex) { const photo = formPhotos[itemIndex]?.[photoIndex]; if (!photo) return; Swal.fire({ title: 'Edit Keterangan Foto', input: 'text', inputLabel: 'Keterangan', inputValue: photo.caption || '', showCancelButton: true, confirmButtonText: 'Simpan', cancelButtonText: 'Batal' }).then((result) => { if (result.isConfirmed) { formPhotos[itemIndex][photoIndex].caption = result.value; } }); } function deletePhoto(itemIndex, photoIndex) { Swal.fire({ title: 'Hapus foto ini?', text: 'Foto akan dihapus dari lampiran.', icon: 'warning', showCancelButton: true, confirmButtonColor: '#d33', confirmButtonText: 'Ya, hapus!' }).then((result) => { if (result.isConfirmed) { formPhotos[itemIndex].splice(photoIndex, 1); renderPhotosInForm(itemIndex); } }); } function renderLampiran(lampiran) { const grid = document.getElementById('lampiranGrid'); const area = document.getElementById('lampiranArea'); if (!grid || !area) return; if (!lampiran || lampiran.length === 0) { area.classList.add('hidden'); return; } area.classList.remove('hidden'); grid.innerHTML = ''; lampiran.forEach((item, idx) => { const div = document.createElement('div'); div.className = 'lampiran-item'; const caption = item.caption || `Lampiran ke-${idx + 1}`; div.innerHTML = `
${caption}
`; grid.appendChild(div); }); }
